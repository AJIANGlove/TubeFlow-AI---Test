# 5. 盘点三个 schema
select n.nspname as schema,
       pg_get_userbyid(n.nspowner) as owner,
       count(c.oid) filter (where c.relkind='r') as tables,
       count(c.oid) filter (where c.relkind='i') as indexes,
       count(c.oid) filter (where c.relkind='S') as sequences,
       count(c.oid) filter (where c.relkind='v') as views,
       count(c.oid) filter (where c.relkind='m') as matviews,
       pg_size_pretty(coalesce(sum(pg_total_relation_size(c.oid))
                      filter (where c.relkind in ('r','m')), 0)) as total_size
from pg_namespace n
left join pg_class c on c.relnamespace = n.oid
where n.nspname in ('audit_272879e_20260829a','audit_272879e_exact_20260829a','staged_cleanup_gate')
group by 1,2 order by 1;

select n.nspname as schema,
       (select count(*) from pg_proc p where p.pronamespace = n.oid) as functions,
       (select count(*) from pg_type t where t.typnamespace = n.oid) as types
from pg_namespace n
where n.nspname in ('audit_272879e_20260829a','audit_272879e_exact_20260829a','staged_cleanup_gate')
order by 1;

-- 每张表：reltuples 与占用（-1 表示从未 ANALYZE，不是行数）
select n.nspname as schema, c.relname as table, c.reltuples::bigint as est_rows,
       pg_size_pretty(pg_total_relation_size(c.oid)) as size
from pg_class c join pg_namespace n on n.oid = c.relnamespace
where n.nspname in ('audit_272879e_20260829a','audit_272879e_exact_20260829a','staged_cleanup_gate')
  and c.relkind = 'r'
order by 1, 2;

-- 真实行数（逐表 count(*) 汇总；只读）
select n.nspname as schema,
       sum((xpath('/row/cnt/text()',
            query_to_xml(format('select count(*) as cnt from %I.%I', n.nspname, c.relname),
                         false, true, '')))[1]::text::bigint) as real_rows,
       count(*) as tables
from pg_class c join pg_namespace n on n.oid = c.relnamespace
where n.nspname in ('audit_272879e_20260829a','audit_272879e_exact_20260829a','staged_cleanup_gate')
  and c.relkind = 'r'
group by 1 order by 1;

-- 有数据的表
select n.nspname as schema, c.relname as table,
       (xpath('/row/cnt/text()',
         query_to_xml(format('select count(*) as cnt from %I.%I', n.nspname, c.relname),
                      false, true, '')))[1]::text::bigint as rows
from pg_class c join pg_namespace n on n.oid = c.relnamespace
where n.nspname in ('audit_272879e_20260829a','audit_272879e_exact_20260829a','staged_cleanup_gate')
  and c.relkind = 'r'
  and (xpath('/row/cnt/text()',
        query_to_xml(format('select count(*) as cnt from %I.%I', n.nspname, c.relname),
                     false, true, '')))[1]::text::bigint > 0
order by 1, 2;
